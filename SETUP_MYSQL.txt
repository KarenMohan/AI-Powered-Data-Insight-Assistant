CREATE DATABASE IF NOT EXISTS ai_data_insight
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE ai_data_insight;


CREATE TABLE IF NOT EXISTS users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX ix_users_username (username)
);

CREATE TABLE IF NOT EXISTS datasets (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    filename VARCHAR(255) NOT NULL,
    csv_data LONGBLOB NOT NULL,
    row_count INT NOT NULL,
    column_count INT NOT NULL,
    uploaded_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX ix_datasets_user_id (user_id),
    INDEX ix_datasets_uploaded_at (uploaded_at),
    CONSTRAINT fk_datasets_user
        FOREIGN KEY (user_id) REFERENCES users(id)
        ON DELETE CASCADE
);

SELECT * from users;

SELECT id, user_id, filename, row_count, column_count, uploaded_at FROM datasets;
